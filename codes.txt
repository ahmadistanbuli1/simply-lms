validations/course.schema.ts
import { z } from "zod";

export const createCourseSchema = z.object({
  title: z
    .string()
    .min(3, { message: "Title must be at least 3 characters." })
    .max(20, { message: "Title is too long." }),
  description: z
    .string()
    .min(10, { message: "Description must be at least 10 characters." })
    .max(1000, { message: "Description cannot exceed 1000 characters." }),
});

export type CreateCourseInput = z.infer<typeof createCourseSchema>;

export const courseDetailsSchema = z.object({
  title: z
    .string()
    .min(3, { error: "Title Must Be At Least 3 Characters" })
    .max(20, { error: "Title Must Be At Least 20 Characters" }),
  description: z
    .string()
    .min(10, { message: "Description must be at least 10 characters." })
    .max(1000, { message: "Description cannot exceed 1000 characters." }),
});

export type CourseDetailsInput = z.infer<typeof courseDetailsSchema>;

export const updateCourseSchema = z.object({
  title: createCourseSchema.shape.title.optional(),
  description: courseDetailsSchema.shape.description.optional(),
  imageUrl: z.string().url({ message: "Invalid image URL." }).optional(),
  isPublished: z.boolean().optional(),
});

export type UpdateCourseInput = z.infer<typeof updateCourseSchema>;

"use client";

import { useState } from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import {
  courseDetailsSchema,
  CourseDetailsInput,
} from "../validations/course.schema";
import { useUpdateCourse } from "../hooks/useUpdateCourse";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import ErrorMessage from "@/components/ErrorMessage";

interface CourseDetailsFormProps {
  initialData: {
    title: string;
    description: string | null;
  };
  courseId: string;
}

export function CourseDetailsForm({
  initialData,
  courseId,
}: CourseDetailsFormProps) {
  const [isEditing, setIsEditing] = useState(false);
  const { mutate: updateCourse, isPending } = useUpdateCourse(courseId);
  const { title, description } = initialData;

  const {
    register,
    handleSubmit,
    reset,

    formState: { errors },
  } = useForm<CourseDetailsInput>({
    resolver: zodResolver(courseDetailsSchema),
    defaultValues: {
      title: title ?? "",
      description: description ?? "",
    },
    values: { title: title ?? "", description: description ?? "" },
    resetOptions: { keepDirtyValues: true },
  });

  const handleToggleEdit = () => {
    if (isEditing) {
      // If Cancel is pressed, the field is restored to its original saved value
      reset({ title: title ?? "", description: description ?? "" });
    }
    setIsEditing((prev) => !prev);
  };

  const onSubmit = (values: CourseDetailsInput) => {
    updateCourse(values, {
      onSuccess: () => {
        setIsEditing(false);
      },
    });
  };

  return (
    <div className="mt-6 border bg-slate-100 rounded-md p-4">
      <div className="font-medium flex items-center justify-between">
        <span>Course Details</span>
        <Button
          type="button"
          onClick={handleToggleEdit}
          variant="ghost"
          size="sm"
          disabled={isPending}
        >
          {isEditing
            ? "Cancel"
            : description
              ? "Edit description"
              : "Add description"}
        </Button>
      </div>

      {!isEditing && (
        <>
          <span className="font-bold text-black">{title}</span>
          <p
            className={`text-sm mt-2 ${
              !description ? "text-slate-500 italic" : "text-slate-700"
            }`}
          >
            {description || "No description provided yet."}
          </p>
        </>
      )}

      {isEditing && (
        <form onSubmit={handleSubmit(onSubmit)} className="space-y-4 mt-4">
          <div className="flex flex-col gap-2">
            <Input {...register("title")} />
            {errors.title?.message && (
              <ErrorMessage msg={errors.title.message} />
            )}
            <textarea
              {...register("description")}
              disabled={isPending}
              rows={4}
              className="w-full p-2.5 text-sm border rounded-md bg-white focus:outline-none focus:ring-2 focus:ring-slate-400 disabled:opacity-50"
              placeholder="e.g. 'This course covers building scalable full-stack applications...'"
            />
            {errors.description?.message && (
              <ErrorMessage msg={errors.description.message} />
            )}
          </div>

          <div className="flex items-center gap-x-2">
            <Button disabled={isPending} type="submit" size="sm">
              {isPending ? "Saving..." : "Save"}
            </Button>
          </div>
        </form>
      )}
    </div>
  );
}

CourseSetupContent.tsx
"use client";

import { useGetCourse } from "../hooks/useGetCourse";
import { CourseDetailsForm } from "./CourseDetailsForm";

interface Props {
  courseId: string; // Not Promise
}

export function CourseSetupContent({ courseId }: Props) {
  const { data: course, isLoading, isError } = useGetCourse(courseId);

  if (isLoading) {
    return (
      <div className="p-6 flex items-center justify-center min-h-100">
        <p className="text-sm text-slate-500 animate-pulse">
          Loading course details...
        </p>
      </div>
    );
  }

  if (isError || !course) {
    return (
      <div className="p-6 flex items-center justify-center min-h-100">
        <p className="text-sm text-red-500 font-medium">
          Failed to load course or access denied.
        </p>
      </div>
    );
  }

  const requiredFields = [course.title, course.description, course.imageUrl];

  const totalFields = requiredFields.length;
  const completedFields = requiredFields.filter(Boolean).length;
  const completionText = `(${completedFields}/${totalFields})`;

  return (
    <div className="p-6 max-w-5xl mx-auto">
      <div className="flex items-center justify-between">
        <div className="flex flex-col gap-y-2">
          <h1 className="text-2xl font-semibold text-slate-900">
            Course Setup: {course.title}
          </h1>
          <span className="text-sm text-slate-600">
            Complete all fields to publish your course {completionText}
          </span>
        </div>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mt-8">
        <div>
          <h2 className="text-xl font-medium text-slate-800">
            Customize your course
          </h2>

          <CourseDetailsForm
            initialData={{
              title: course.title,
              description: course.description,
            }}
            courseId={course.id}
          />
        </div>

        <div>
          <h2 className="text-xl font-medium text-slate-800">
            Course chapters
          </h2>
          <div className="mt-6 border bg-slate-100 rounded-md p-4 text-sm text-slate-500">
            Chapters management will be implemented here next.
          </div>
        </div>
      </div>
    </div>
  );
}
